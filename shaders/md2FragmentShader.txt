#version 330 core

uniform sampler2D tex;
uniform vec3 lightDirection;

in vec3 norm;
in vec2 uv;

out vec4 out_Color;

void main(void){
	float diffuse = max(dot(normalize(norm), normalize(lightDirection)), 0.0);
	float light = 0.35 + 0.65 * diffuse;
	vec4 color = texture(tex, uv);

	out_Color = vec4(color.rgb * light, color.a);
}
